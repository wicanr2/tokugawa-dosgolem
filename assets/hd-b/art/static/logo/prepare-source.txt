"""保留 Family Soft 字樣與原版位置，校準 OP1 的手繪字標。"""
import argparse
import hashlib
import json
import os
from pathlib import Path

from PIL import Image, ImageDraw, __version__ as pillow_version


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    assert Path('/.dockerenv').is_file() and os.getuid() == 1000 and pillow_version == '9.4.0'
    parser = argparse.ArgumentParser()
    parser.add_argument('--integrate',action='store_true')
    args = parser.parse_args()
    project,root = Path('/project'),Path('/output')
    assert root.stat().st_uid == root.stat().st_gid == 1000
    catalog = project/'workplace/hd-prototype/catalog'
    if args.integrate:
        manifest_path,review_path = root/'draft-manifest.json',root/'review.json'
        m,review = [json.loads(p.read_text()) for p in [manifest_path,review_path]]
        assert review['verified'] and review['draft_manifest_sha256'] == sha(manifest_path)
        assert m['tool_sha256'] == sha(Path(__file__)) == sha(root/'prepare-source.txt')
        r = m['record']
        assert sha(root/r['prepared']) == r['asset_sha256']
        source = r['source']
        planes = ','.join(json.dumps(s) for s in source['plane_sha256'])
        rect = ','.join(map(str,r['paint_rect']))
        code = '// Code generated by tools/hd-logo-prepare.py; DO NOT EDIT.\npackage main\nimport "image"\n'
        code += 'var hdGraphicsLogoProfile = hdGraphicsProfile{Source:graphicsSourceProfile{Name:"OP1.OZM",Width:640,Height:400,Planes:[]string{'+planes+'}},'
        code += 'IndicesSHA:'+json.dumps(source['indices_sha256'])+',Asset:'+json.dumps(r['asset'])+',AssetSHA:'+json.dumps(r['asset_sha256'])+',PaintRect:image.Rect('+rect+')}\n'
        target = project/'tools/hd-graphics-logo-profile.go'
        assert not target.exists() and target.parent.stat().st_uid == 1000
        target.write_text(code)
        target = root/'profiles.json';assert not target.exists()
        target.write_text(json.dumps({**m,'classification':'reviewed original publisher wordmark geometry and common opaque consumer profile',
                                     'review_sha256':sha(review_path),'draft_manifest_sha256':sha(manifest_path)},ensure_ascii=False,indent=2)+'\n')
        print(json.dumps({'approved_logo':'OP1.OZM'}))
        return
    source = next(r for r in json.loads((catalog/'decoded/inventory.json').read_text())['records'] if r['name']=='OP1.OZM')
    idx_path = catalog/'decoded/OP1.OZM.idx';assert sha(idx_path)==source['indices_sha256']
    data = idx_path.read_bytes()
    mask = Image.frombytes('L',(640,400),bytes(255 if v not in [0,8] else 0 for v in data))
    rect = mask.getbbox();assert rect and 200 <= rect[2]-rect[0] <= 260 and 40 <= rect[3]-rect[1] <= 80
    generation = json.loads((catalog/'art/logo-generation-v1.json').read_text())
    raw_path = catalog/'art'/generation['asset'];assert sha(raw_path)==generation['sha256']
    raw = Image.open(raw_path).convert('RGB')
    bright = raw.convert('L').point(lambda v:255 if v>40 else 0).getbbox();assert bright
    crop = (max(0,bright[0]-3),max(0,bright[1]-3),min(raw.width,bright[2]+3),min(raw.height,bright[3]+3))
    out = Image.new('RGB',(1280,800),(0,0,0))
    out.paste(raw.crop(crop).resize(((rect[2]-rect[0])*2,(rect[3]-rect[1])*2),Image.Resampling.LANCZOS),(rect[0]*2,rect[1]*2))
    dest = root/'OP1-hd-calibrated-v1.png';assert not dest.exists();out.save(dest)
    sheet = Image.new('RGB',(1280,430),(222,216,197));ImageDraw.Draw(sheet).text((8,8),'OP1.OZM original / calibrated HD',fill=(0,0,0))
    original = Image.open(catalog/'graphics/OP1.OZM-original.png').convert('RGB').resize((600,375),Image.Resampling.NEAREST)
    sheet.paste(original,(10,32));sheet.paste(out.resize((600,375),Image.Resampling.LANCZOS),(650,32))
    review = root/'OP1.OZM-review.png';sheet.save(review)
    record = {'name':'OP1.OZM','source':source,'raw_asset':generation['asset'],'raw_asset_sha256':generation['sha256'],
              'prompt':generation['prompt'],'prompt_sha256':generation['prompt_sha256'],'raw_crop':list(crop),
              'paint_rect':list(rect),'prepared':dest.name,'asset':'static/'+dest.name,'asset_sha256':sha(dest)}
    target = root/'draft-manifest.json';assert not target.exists()
    target.write_text(json.dumps({'classification':'publisher logo geometry draft; original lettering retained',
                                 'tool_sha256':sha(Path(__file__)),'record':record,'review':review.name,'review_sha256':sha(review)},ensure_ascii=False,indent=2)+'\n')
    (root/'prepare-source.txt').write_bytes(Path(__file__).read_bytes())
    print(json.dumps({'logo_bounds':list(rect),'raw_crop':list(crop),'sha256':sha(dest)}))


if __name__ == '__main__':
    main()
